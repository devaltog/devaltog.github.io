// PASTE THIS into Firebase Console → Firestore Database → Rules → Publish.
// Anyone can READ (so your public site works for visitors).
// Only a signed-in trainer account can WRITE (add/edit/delete).
// Since you never give out a sign-up form, only accounts you create in
// Firebase Console → Authentication can sign in and write.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
